+static IRAM_ATTR bool test_i2c_stretch_callback(i2c_slave_dev_handle_t channel, const i2c_slave_stretch_event_data_t *edata, void *user_data)
+{
+    uint8_t data_wr[LENGTH] = { 0 };
+    if (edata->stretch_cause == I2C_SLAVE_STRETCH_CAUSE_ADDRESS_MATCH) {
+        uint32_t fifo_size;
+        i2c_slave_get_tx_empty_size(channel, &fifo_size);
+        if (fifo_size == SOC_I2C_FIFO_LEN) {
+            i2c_slave_write_one_shot(channel, data_wr, LENGTH);
+        }
+    }
+    return true;
+}
